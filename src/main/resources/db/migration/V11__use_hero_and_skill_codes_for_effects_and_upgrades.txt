-- =========================================================
-- 1. Hero skill effects: replace skill_id with stable hero_code + skill_code
-- =========================================================

ALTER TABLE hero_skill_effects
    ADD COLUMN hero_code VARCHAR(50) NULL AFTER id,
    ADD COLUMN skill_code VARCHAR(50) NULL AFTER hero_code;

UPDATE hero_skill_effects effect_row
JOIN hero_skill_definitions skill
    ON skill.id = effect_row.skill_id
SET effect_row.hero_code = skill.hero_code,
    effect_row.skill_code = skill.skill_code;

ALTER TABLE hero_skill_effects
    MODIFY COLUMN hero_code VARCHAR(50) NOT NULL,
    MODIFY COLUMN skill_code VARCHAR(50) NOT NULL,
    DROP FOREIGN KEY fk_hero_skill_effect_definition,
    DROP INDEX uk_hero_skill_effect_order,
    DROP COLUMN skill_id,
    ADD CONSTRAINT uk_hero_skill_effect_code_order
        UNIQUE (hero_code, skill_code, effect_order),
    ADD INDEX idx_hero_skill_effect_lookup
        (hero_code, skill_code, enabled);


-- =========================================================
-- 2. Hero skill upgrade configs: replace skill_id with stable codes
-- =========================================================

ALTER TABLE hero_skill_upgrade_configs
    ADD COLUMN hero_code VARCHAR(50) NULL AFTER id,
    ADD COLUMN skill_code VARCHAR(50) NULL AFTER hero_code;

UPDATE hero_skill_upgrade_configs upgrade_config
JOIN hero_skill_definitions skill
    ON skill.id = upgrade_config.skill_id
SET upgrade_config.hero_code = skill.hero_code,
    upgrade_config.skill_code = skill.skill_code;

ALTER TABLE hero_skill_upgrade_configs
    MODIFY COLUMN hero_code VARCHAR(50) NOT NULL,
    MODIFY COLUMN skill_code VARCHAR(50) NOT NULL,
    DROP FOREIGN KEY fk_hero_skill_upgrade_config_skill,
    DROP INDEX uk_hero_skill_upgrade_config,
    DROP INDEX idx_hero_skill_upgrade_lookup,
    DROP COLUMN skill_id,
    ADD CONSTRAINT uk_hero_skill_upgrade_code_level
        UNIQUE (hero_code, skill_code, target_skill_level),
    ADD INDEX idx_hero_skill_upgrade_lookup
        (hero_code, skill_code, target_skill_level, enabled);
